<script>

    // 题目路线数据

    const data = [

        {

            q:"下列不属于国家安全范畴的是？",

            opts:["网络安全","家庭聚餐安全","生态安全"],

            correct:1

        },

        {

            q:"发现有人偷拍军事禁区，你应该？",

            opts:["假装没看见","立即报警","拍照转发朋友圈"],

            correct:1

        },

        {

            q:"向境外提供国家秘密会构成？",

            opts:["合法交流","危害国家安全罪","正常学术往来"],

            correct:1

        },

        {

            q:"维护国家安全，公民应当？",

            opts:["遵守宪法法律","随意泄露信息","拒绝配合调查"],

            correct:0

        }

    ];


    let step = 0;

    let pathPos = [0, -90, -180];

    let history = [];

    let audioOn = true;


    // 游戏控制

    function openGame(){

        document.getElementById("gameModal").style.display="block";

        playBgm();

        render();

    }

    function closeGame(){

        document.getElementById("gameModal").style.display="none";

        document.getElementById("bgm").pause();

    }

    function playBgm(){

        document.getElementById("bgm").play().catch(e=>{});

    }

    function toggleAudio(){

        let a = document.getElementById("bgm");

        let btn = document.getElementById("audioBtn");

        if(audioOn){

            a.pause();

            btn.textContent="开启音乐";

        }else{

            a.play();

            btn.textContent="关闭音乐";

        }

        audioOn=!audioOn;

    }


    // 渲染当前题目与道路

    function render(){

        let d = data[step];

        document.getElementById("questionText").textContent = d.q;

        let road = document.getElementById("road");

        road.innerHTML = "";

        d.opts.forEach((opt,i)=>{

            let p = document.createElement("div");

            p.className="path";

            p.onclick=()=>choose(i);

            road.appendChild(p);

        });

        document.getElementById("people").style.left="calc(50% - 20px)";

        document.getElementById("backBtn").style.display = step>0 ? "block" : "none";

    }


    // 选择答案

    function choose(idx){

        if(step>=data.length) return;

        history.push(idx);

        movePeople(idx);

        setTimeout(()=>{

            step++;

            if(step>=data.length){

                showResult();

            }else{

                render();

            }

        },900);

    }


    // 人物移动

    function movePeople(idx){

        let p = document.getElementById("people");

        p.style.left = `calc(50% + ${pathPos[idx]}px)`;

        document.getElementById("questionText").textContent="正在前行...";

    }


    // 返回上一步

    function goBack(){

        if(step<=0) return;

        history.pop();

        step--;

        render();

    }


    // 结局判断

    function showResult(){

        document.querySelector(".road-box").style.display="none";

        document.getElementById("backBtn").style.display="none";

        let resBox = document.getElementById("resultBox");

        let title = document.getElementById("resultTitle");

        let content = document.getElementById("resultContent");

        resBox.style.display="flex";


        let ok = history.every((s,i)=>s===data[i].correct);

        if(ok){

            title.textContent="恭喜你！你是合格守法好公民！";

            content.textContent="你的选择全部正确，国家安全知识掌握扎实，感谢你为维护国家安全贡献力量！";

        }else{

            title.textContent="选择有误，请注意法律风险";

            content.textContent="部分选择不符合国家安全法律要求，错误行为可能危害国家安全，需承担法律责任，请认真学习相关法律知识。";

        }

    }


    function restartGame(){

        step=0;

        history=[];

        document.querySelector(".road-box").style.display="flex";

        document.getElementById("resultBox").style.display="none";

        render();

    }

</script>
